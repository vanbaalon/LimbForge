cmake_minimum_required(VERSION 3.20)
set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum supported macOS version")
project(LimbForge VERSION 0.1.0 LANGUAGES CXX OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "LimbForge's GPU backend requires macOS and Metal")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
file(READ include/limbforge/core.hpp CORE_SOURCE)
string(REPLACE "#pragma once" "" CORE_SOURCE "${CORE_SOURCE}")
file(READ src/kernels.metal KERNEL_SOURCE)
configure_file(src/shader_source.hpp.in generated/shader_source.hpp @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS include/limbforge/core.hpp src/kernels.metal)
add_library(limbforge src/engine.mm)
add_library(LimbForge::limbforge ALIAS limbforge)
target_compile_features(limbforge PUBLIC cxx_std_17)
target_include_directories(limbforge PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(limbforge PRIVATE -fobjc-arc -Wall -Wextra)
target_link_libraries(limbforge PUBLIC "-framework Metal" "-framework Foundation")
include(GNUInstallDirs)
install(TARGETS limbforge EXPORT LimbForgeTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT LimbForgeTargets NAMESPACE LimbForge:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/LimbForge)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/LimbForgeConfig.cmake.in
  "${CMAKE_CURRENT_BINARY_DIR}/LimbForgeConfig.cmake"
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/LimbForge)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/LimbForgeConfigVersion.cmake"
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/LimbForgeConfig.cmake"
  "${CMAKE_CURRENT_BINARY_DIR}/LimbForgeConfigVersion.cmake"
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/LimbForge)
include(CTest)
option(LIMBFORGE_BUILD_BENCHMARK "Build the MPFR comparison benchmark" ON)
option(LIMBFORGE_BUILD_EXAMPLES "Build the division, resident, and norm examples" ON)
option(LIMBFORGE_ENABLE_GPU_TESTS "Run tests that require a Metal GPU" ON)
option(LIMBFORGE_BUILD_BAXTER_EXAMPLE "Build the BSolver4D comparison" OFF)
set(LIMBFORGE_BAXTER_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../BSolver4D/cpp" CACHE PATH "Directory containing BSolver4D baxter.hpp")
if(BUILD_TESTING OR LIMBFORGE_BUILD_BENCHMARK OR LIMBFORGE_BUILD_BAXTER_EXAMPLE OR LIMBFORGE_BUILD_EXAMPLES)
  find_path(MPFR_INCLUDE_DIR mpfr.h HINTS /opt/homebrew/opt/mpfr/include REQUIRED)
  find_path(GMP_INCLUDE_DIR gmp.h HINTS /opt/homebrew/opt/gmp/include REQUIRED)
  find_library(MPFR_LIBRARY mpfr HINTS /opt/homebrew/opt/mpfr/lib REQUIRED)
  find_library(GMP_LIBRARY gmp HINTS /opt/homebrew/opt/gmp/lib REQUIRED)
  add_library(limbforge_reference INTERFACE)
  target_include_directories(limbforge_reference INTERFACE tests ${MPFR_INCLUDE_DIR} ${GMP_INCLUDE_DIR})
  target_link_libraries(limbforge_reference INTERFACE limbforge ${MPFR_LIBRARY} ${GMP_LIBRARY})
  # Optional MPC (mpc_t bridge checks and benchmark); the bridge header itself needs only <mpc.h>.
  find_path(MPC_INCLUDE_DIR mpc.h HINTS /opt/homebrew/opt/libmpc/include)
  find_library(MPC_LIBRARY mpc HINTS /opt/homebrew/opt/libmpc/lib)
  function(limbforge_use_mpc target)
    if(MPC_INCLUDE_DIR AND MPC_LIBRARY)
      target_include_directories(${target} PRIVATE ${MPC_INCLUDE_DIR})
      target_link_libraries(${target} PRIVATE ${MPC_LIBRARY})
      target_compile_definitions(${target} PRIVATE LIMBFORGE_TEST_MPC)
    endif()
  endfunction()
  if(NOT (MPC_INCLUDE_DIR AND MPC_LIBRARY))
    message(STATUS "MPC not found: mpc_t bridge checks are skipped")
  endif()
endif()
if(LIMBFORGE_BUILD_BAXTER_EXAMPLE)
  if(NOT EXISTS "${LIMBFORGE_BAXTER_SOURCE_DIR}/baxter.hpp")
    message(FATAL_ERROR "Set LIMBFORGE_BAXTER_SOURCE_DIR to a BSolver4D/cpp checkout")
  endif()
  find_path(BOOST_INCLUDE_DIR boost/multiprecision/mpfr.hpp HINTS /opt/homebrew/opt/boost/include REQUIRED)
  add_executable(baxter_batch benchmarks/baxter_batch.cpp)
  target_include_directories(baxter_batch PRIVATE ${LIMBFORGE_BAXTER_SOURCE_DIR} ${BOOST_INCLUDE_DIR})
  target_link_libraries(baxter_batch PRIVATE limbforge_reference)
  if(BUILD_TESTING AND LIMBFORGE_ENABLE_GPU_TESTS)
    add_test(NAME baxter_batch COMMAND baxter_batch 4 600)
  endif()
endif()
if(BUILD_TESTING)
  add_executable(test_limbforge tests/test_arithmetic.cpp)
  target_link_libraries(test_limbforge PRIVATE limbforge_reference)
  add_test(NAME arithmetic_cpu COMMAND test_limbforge --cpu-only)
  add_executable(test_limbforge_bridge tests/test_bridge.cpp)
  target_link_libraries(test_limbforge_bridge PRIVATE limbforge_reference)
  limbforge_use_mpc(test_limbforge_bridge)
  add_test(NAME bridge_cpu COMMAND test_limbforge_bridge)
  add_executable(test_limbforge_fused tests/test_fused.cpp)
  target_link_libraries(test_limbforge_fused PRIVATE limbforge_reference)
  add_test(NAME fused_cpu COMMAND test_limbforge_fused --cpu-only)
  if(LIMBFORGE_ENABLE_GPU_TESTS)
    add_test(NAME fused_gpu COMMAND test_limbforge_fused)
  endif()
  if(LIMBFORGE_ENABLE_GPU_TESTS)
    add_test(NAME arithmetic_gpu COMMAND test_limbforge)
    # Diagnostic only (plan A2): not a CTest; compares GPU product formulations with the CPU.
    add_executable(gpu_codegen_probe tests/gpu_codegen_probe.mm)
    target_include_directories(gpu_codegen_probe PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_options(gpu_codegen_probe PRIVATE -fobjc-arc -Wall -Wextra)
    target_compile_definitions(gpu_codegen_probe PRIVATE LIMBFORGE_PROBE_VARIANTS="${CMAKE_CURRENT_SOURCE_DIR}/tests/probe_variants.hpp")
    target_link_libraries(gpu_codegen_probe PRIVATE limbforge_reference)
    # Diagnostic only (plan L1): TensorOps integer matmul exactness, overflow and throughput.
    add_executable(tensorops_probe tests/tensorops_probe.mm)
    target_compile_options(tensorops_probe PRIVATE -fobjc-arc -Wall -Wextra)
    target_link_libraries(tensorops_probe PRIVATE "-framework Metal" "-framework Foundation")
    add_executable(test_limbforge_buffers tests/test_buffers.cpp)
    target_link_libraries(test_limbforge_buffers PRIVATE limbforge_reference)
    add_test(NAME resident_buffers COMMAND test_limbforge_buffers)
    add_executable(test_limbforge_reduction tests/test_reduction.cpp)
    target_link_libraries(test_limbforge_reduction PRIVATE limbforge_reference)
    add_test(NAME tree_reduction COMMAND test_limbforge_reduction)
  endif()
endif()
if(LIMBFORGE_BUILD_EXAMPLES)
  add_executable(limbforge_division examples/division.cpp)
  target_link_libraries(limbforge_division PRIVATE limbforge_reference)
  add_executable(limbforge_resident examples/resident.cpp)
  target_link_libraries(limbforge_resident PRIVATE limbforge_reference)
  add_executable(limbforge_reduction examples/reduction.cpp)
  target_link_libraries(limbforge_reduction PRIVATE limbforge_reference)
endif()
if(LIMBFORGE_BUILD_BENCHMARK)
  add_executable(benchmark_limbforge benchmarks/benchmark.cpp)
  target_link_libraries(benchmark_limbforge PRIVATE limbforge_reference)
  add_executable(reduction_limbforge benchmarks/reduction.cpp)
  target_link_libraries(reduction_limbforge PRIVATE limbforge_reference)
  add_executable(square_limbforge benchmarks/square.cpp)
  target_link_libraries(square_limbforge PRIVATE limbforge_reference)
  add_executable(kernel_sweep benchmarks/kernel_sweep.cpp)
  target_link_libraries(kernel_sweep PRIVATE limbforge_reference)
  add_executable(bridge_limbforge benchmarks/bridge.cpp)
  target_link_libraries(bridge_limbforge PRIVATE limbforge_reference)
  limbforge_use_mpc(bridge_limbforge)
  add_executable(tune_limbforge benchmarks/tune.cpp)
  target_link_libraries(tune_limbforge PRIVATE limbforge_reference)
  add_executable(layout_limbforge benchmarks/layout.mm)
  target_include_directories(layout_limbforge PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_options(layout_limbforge PRIVATE -fobjc-arc -Wall -Wextra)
  target_link_libraries(layout_limbforge PRIVATE limbforge_reference)
endif()
