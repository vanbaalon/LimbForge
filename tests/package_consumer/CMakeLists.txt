cmake_minimum_required(VERSION 3.20)
project(WolfNumPackageConsumer LANGUAGES CXX)
find_package(LimbForge ${LF_REQUIRED_VERSION} CONFIG REQUIRED)
if(NOT LimbForge_VERSION STREQUAL LF_EXPECTED_VERSION OR NOT LimbForge_API_VERSION STREQUAL LF_EXPECTED_API)
  message(FATAL_ERROR "Installed package reports an unexpected version/API")
endif()
add_executable(package_consumer main.cpp)
target_link_libraries(package_consumer PRIVATE LimbForge::limbforge)
target_compile_definitions(package_consumer PRIVATE LF_EXPECTED_VERSION="${LF_EXPECTED_VERSION}")
if(LF_FOREIGN_HEADERS)
  # Simulate one newer header with the original archive, including in unoptimized builds.
  get_target_property(includes LimbForge::limbforge INTERFACE_INCLUDE_DIRECTORIES)
  list(GET includes 0 include_dir)
  file(READ "${include_dir}/limbforge/version.hpp" version_header)
  string(REGEX MATCH "#define LIMBFORGE_VERSION_PATCH ([0-9]+)" unused "${version_header}")
  math(EXPR newer_patch "${CMAKE_MATCH_1}+1")
  string(REGEX REPLACE "#define LIMBFORGE_VERSION_PATCH [0-9]+" "#define LIMBFORGE_VERSION_PATCH ${newer_patch}" version_header "${version_header}")
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/foreign/limbforge")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/foreign/limbforge/version.hpp" "${version_header}")
  target_include_directories(package_consumer BEFORE PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/foreign")
  target_compile_definitions(package_consumer PRIVATE LF_EXPECT_MISMATCH=1)
endif()
